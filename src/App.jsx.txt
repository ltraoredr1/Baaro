import React, { useEffect, useState } from "react";
import { api } from "./api.js";

const INK = "#12261F";
const TEAL = "#0E6E5C";
const CORAL = "#E85C3F";
const PAPER = "#F6F8F6";

function LoginScreen({ onLoggedIn }) {
  const [email, setEmail] = useState("admin@test.fr");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  const submit = async (e) => {
    e.preventDefault();
    setError("");
    setLoading(true);
    try {
      const result = await api.login(email, password);
      if (result.role !== "ADMIN") {
        throw new Error("Ce compte n'a pas les droits d'administration");
      }
      onLoggedIn();
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div style={{ minHeight: "100vh", display: "flex", alignItems: "center", justifyContent: "center", background: PAPER, fontFamily: "system-ui, sans-serif" }}>
      <form onSubmit={submit} style={{ background: "#fff", padding: 32, borderRadius: 16, border: "1px solid #E2E9E5", width: 320 }}>
        <h1 style={{ fontSize: 18, fontWeight: 700, color: INK, marginBottom: 4 }}>Administration</h1>
        <p style={{ fontSize: 12.5, color: "#5F6D68", marginBottom: 20 }}>Réservé aux comptes ADMIN</p>

        <label style={{ display: "block", fontSize: 12, fontWeight: 600, color: "#425950", marginBottom: 5 }}>E-mail</label>
        <input value={email} onChange={(e) => setEmail(e.target.value)} type="email" required style={inputStyle} />

        <label style={{ display: "block", fontSize: 12, fontWeight: 600, color: "#425950", margin: "12px 0 5px" }}>Mot de passe</label>
        <input value={password} onChange={(e) => setPassword(e.target.value)} type="password" required style={inputStyle} />

        {error && <p style={{ color: CORAL, fontSize: 12, marginTop: 10 }}>{error}</p>}

        <button type="submit" disabled={loading} style={{
          width: "100%", marginTop: 18, padding: "11px", borderRadius: 10, border: "none",
          background: INK, color: "#fff", fontWeight: 600, fontSize: 13.5, cursor: "pointer",
          opacity: loading ? 0.6 : 1,
        }}>
          {loading ? "Connexion..." : "Se connecter"}
        </button>
      </form>
    </div>
  );
}

const inputStyle = {
  width: "100%", boxSizing: "border-box", padding: "9px 12px", borderRadius: 9,
  border: "1px solid #DCE5E1", fontSize: 13.5,
};

function VerificationQueue() {
  const [pending, setPending] = useState(null);
  const [error, setError] = useState("");
  const [rejectingId, setRejectingId] = useState(null);
  const [reason, setReason] = useState("");
  const [actionLoading, setActionLoading] = useState(null);

  const load = async () => {
    try {
      setPending(await api.pendingVerifications());
    } catch (err) {
      setError(err.message);
    }
  };

  useEffect(() => {
    load();
  }, []);

  const viewDocument = async (doctorId) => {
    try {
      const blob = await api.documentBlob(doctorId);
      const objectUrl = URL.createObjectURL(blob);
      window.open(objectUrl, "_blank", "noopener,noreferrer");
      // Révoquée après un délai large plutôt qu'immédiatement : le nouvel
      // onglet a besoin que l'URL reste valide le temps de charger le blob.
      setTimeout(() => URL.revokeObjectURL(objectUrl), 60_000);
    } catch (err) {
      setError(err.message);
    }
  };

  const approve = async (doctorId) => {
    setActionLoading(doctorId);
    try {
      await api.approve(doctorId);
      await load();
    } catch (err) {
      setError(err.message);
    } finally {
      setActionLoading(null);
    }
  };

  const reject = async (doctorId) => {
    if (reason.trim().length < 3) return;
    setActionLoading(doctorId);
    try {
      await api.reject(doctorId, reason);
      setRejectingId(null);
      setReason("");
      await load();
    } catch (err) {
      setError(err.message);
    } finally {
      setActionLoading(null);
    }
  };

  return (
    <div style={{ maxWidth: 760, margin: "0 auto", padding: "40px 20px" }}>
      <h1 style={{ fontSize: 20, fontWeight: 700, color: INK, marginBottom: 4 }}>Vérifications en attente</h1>
      <p style={{ fontSize: 13, color: "#5F6D68", marginBottom: 24 }}>
        Un médecin n'apparaît dans la recherche patient qu'après approbation.
      </p>

      {error && (
        <div style={{ background: "#FBEAE5", border: "1px solid #F0C4B7", borderRadius: 10, padding: "10px 14px", marginBottom: 16, color: "#9A3A24", fontSize: 13 }}>
          {error}
        </div>
      )}

      {pending === null && <p style={{ color: "#5F6D68", fontSize: 13 }}>Chargement...</p>}
      {pending?.length === 0 && <p style={{ color: "#5F6D68", fontSize: 13 }}>Aucune demande en attente.</p>}

      {pending?.map((doctor) => (
        <div key={doctor.id} style={{ background: "#fff", border: "1px solid #E2E9E5", borderRadius: 14, padding: 18, marginBottom: 12 }}>
          <div style={{ display: "flex", justifyContent: "space-between", alignItems: "flex-start", marginBottom: 10 }}>
            <div>
              <div style={{ fontWeight: 700, color: INK, fontSize: 14.5 }}>{doctor.specialty}</div>
              <div style={{ fontSize: 12.5, color: "#5F6D68" }}>{doctor.user?.email} · {doctor.user?.country}</div>
              <div style={{ fontSize: 11.5, color: "#8B9A94", marginTop: 3 }}>
                Licence {doctor.licenseNumber} · soumis le{" "}
                {doctor.verificationSubmittedAt ? new Date(doctor.verificationSubmittedAt).toLocaleDateString("fr-FR") : "—"}
              </div>
            </div>
            <button onClick={() => viewDocument(doctor.id)} style={{
              fontSize: 12, fontWeight: 600, color: TEAL, background: "#EAF4F1", border: "none",
              borderRadius: 999, padding: "7px 13px", cursor: "pointer", flexShrink: 0,
            }}>
              Voir le document
            </button>
          </div>

          {rejectingId === doctor.id ? (
            <div style={{ display: "flex", gap: 8, marginTop: 10 }}>
              <input
                value={reason}
                onChange={(e) => setReason(e.target.value)}
                placeholder="Motif du rejet (visible par le médecin)"
                style={{ ...inputStyle, flex: 1 }}
              />
              <button onClick={() => reject(doctor.id)} disabled={actionLoading === doctor.id} style={btnStyle(CORAL)}>
                Confirmer
              </button>
              <button onClick={() => { setRejectingId(null); setReason(""); }} style={btnStyle("#DCE5E1", "#425950")}>
                Annuler
              </button>
            </div>
          ) : (
            <div style={{ display: "flex", gap: 8 }}>
              <button onClick={() => approve(doctor.id)} disabled={actionLoading === doctor.id} style={btnStyle(TEAL)}>
                {actionLoading === doctor.id ? "..." : "Approuver"}
              </button>
              <button onClick={() => setRejectingId(doctor.id)} style={btnStyle("#fff", CORAL, `1px solid ${CORAL}`)}>
                Rejeter
              </button>
            </div>
          )}
        </div>
      ))}
    </div>
  );
}

function btnStyle(bg, color = "#fff", border = "none") {
  return {
    padding: "8px 16px", borderRadius: 9, border, background: bg, color,
    fontWeight: 600, fontSize: 12.5, cursor: "pointer",
  };
}

export default function App() {
  const [loggedIn, setLoggedIn] = useState(null); // null = vérification en cours
  const [checkError, setCheckError] = useState("");

  useEffect(() => {
    api.me()
      .then((user) => setLoggedIn(user?.role === "ADMIN"))
      .catch(() => setLoggedIn(false));
  }, []);

  const logout = async () => {
    try {
      await api.logout();
    } catch (err) {
      setCheckError(err.message);
    } finally {
      setLoggedIn(false);
    }
  };

  if (loggedIn === null) {
    return (
      <div style={{ minHeight: "100vh", display: "flex", alignItems: "center", justifyContent: "center", background: PAPER, fontFamily: "system-ui, sans-serif", color: "#5F6D68", fontSize: 13 }}>
        Vérification de la session...
      </div>
    );
  }

  if (!loggedIn) {
    return <LoginScreen onLoggedIn={() => setLoggedIn(true)} />;
  }

  return (
    <div style={{ minHeight: "100vh", background: PAPER, fontFamily: "system-ui, sans-serif" }}>
      <div style={{ background: "#fff", borderBottom: "1px solid #E2E9E5", padding: "14px 20px", display: "flex", justifyContent: "space-between", alignItems: "center" }}>
        <span style={{ fontWeight: 700, color: INK, fontSize: 14 }}>Télémédecine — Admin</span>
        <button
          onClick={logout}
          style={{ background: "none", border: "none", color: "#5F6D68", fontSize: 12.5, cursor: "pointer" }}
        >
          Déconnexion
        </button>
      </div>
      {checkError && (
        <div style={{ background: "#FBEAE5", border: "1px solid #F0C4B7", borderRadius: 10, padding: "10px 14px", margin: "16px 20px", color: "#9A3A24", fontSize: 13 }}>
          {checkError}
        </div>
      )}
      <VerificationQueue />
    </div>
  );
}
