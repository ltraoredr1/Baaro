// 5. PAGE INVITATION (optionnel, crée src/pages/InvitePage.jsx)
import { useEffect } from 'react'
import { useParams, useNavigate } from 'react-router-dom'

export function InvitePage() {
  const { code } = useParams()
  const navigate = useNavigate()
  const { joinViaCode } = useCommunityExtras(session?.user?.id)
  
  useEffect(() => {
    if(!code) return
    joinViaCode(code).then(groupId => {
      navigate('/community?group=' + groupId)
    }).catch(err => alert(err.message))
  }, [code])
  
  return <div className="p-8 text-center">Rejoins le groupe... Code: {code}</div>
}
